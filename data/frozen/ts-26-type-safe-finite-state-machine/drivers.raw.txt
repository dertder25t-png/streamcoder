probe("initial state", async () => {
  const machine = createStateMachine({
    initial: "start",
    transitions: {
      start: {
        next: { target: "end" }
      }
    }
  });
  return machine.getState();
});

  const machine = createStateMachine({
    initial: "start",
    transitions: {

      }
    }
  });
  return machine.can("next");
