- Returns true only when payload is a non-null plain object with type === "payment.settled", a non-empty string id, and a non-negative integer amountCents.
- Returns false for null, primitives, arrays, fractional amounts, or negative amounts without throwing.

```ts
test("returns true for valid PaymentWebhook", () => {
  expect(isPaymentWebhook({ type: "payment.settled", id: "123", amountCents: 100 })).toBe(true);
});

test("returns false for null", () => {
  expect(isPaymentWebhook(null)).toBe(false);
});

test("returns false for primitive values", () => {
  expect(isPaymentWebhook("string")).toBe(false);
  expect(isPaymentWebhook(123)).toBe(false);
  expect(isPaymentWebhook(true)).toBe(false);
});

test("returns false for arrays", () => {
  expect(isPaymentWebhook(["payment.settled", "123", 100])).toBe(false);
});

test("returns false for fractional amountCents", () => {
  expect(isPaymentWebhook({ type: "payment.settled", id: "123", amountCents: 100.5 })).toBe(false);
});

test("returns false for negative amountCents", () => {
  expect(isPaymentWebhook({ type: "payment.settled", id: "123", amountCents: -100 })).toBe(false);
});
```
