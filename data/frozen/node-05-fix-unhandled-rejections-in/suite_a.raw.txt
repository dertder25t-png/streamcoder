- Resolves to a map of path → string for every existing readable file, silently omitting paths that fail specifically with ENOENT.
- Re-throws non-ENOENT errors (such as EISDIR when a directory path is supplied).
- Never emits unhandled promise rejection warnings.

```ts
import { promises as fs } from "node:fs";
import { readExistingFiles } from "./readExistingFiles"; // Adjust the import path as necessary

test("reads existing files and omits ENOENT errors", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const filePath1 = path.join(dir, "file1.txt");
  const filePath2 = path.join(dir, "file2.txt");

  await fs.promises.writeFile(filePath1, "content1");
  await fs.promises.unlink(filePath2); // Remove file2 to simulate ENOENT

  const result = await readExistingFiles([filePath1, filePath2]);
  expect(result).toEqual(new Map([[filePath1, "content1"]]));
});

test("re-throws non-ENOENT errors", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const filePath = path.join(dir, "file.txt");

  await fs.promises.mkdir(filePath); // Create a directory to simulate EISDIR

  await expect(readExistingFiles([filePath])).rejects.toThrow("EISDIR");
});

test("does not emit unhandled promise rejection warnings", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const filePath = path.join(dir, "file.txt");

  await fs.promises.unlink(filePath); // Remove file to simulate ENOENT

  const spy = vi.spyOn(console, "error");
  await readExistingFiles([filePath]);
  expect(spy).not.toHaveBeenCalled();
});
```
