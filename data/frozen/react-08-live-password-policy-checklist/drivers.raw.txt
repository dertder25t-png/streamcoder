probe("empty password", async () => {
  render(<PasswordPolicyMeter onValidChange={() => {}} />);
  await waitFor(() => {
    expect(screen.getByRole("checkbox", { name: "at least 10 characters" })).toHaveAttribute("data-passed", "false");
    expect(screen.getByRole("checkbox", { name: "an uppercase letter" })).toHaveAttribute("data-passed", "false");
    expect(screen.getByRole("checkbox", { name: "a digit" })).toHaveAttribute("data-passed", "false");
    expect(screen.getByRole("checkbox", { name: "a symbol" })).toHaveAttribute("data-passed", "false");
  });
  return snap();
});
