import { promises as fs } from 'node:fs';
import { join, basename } from 'node:path';

interface FileStats {
  count: number;
  bytes: number;
}

async function summarizeByExtension(dirPath: string): Promise<Record<string, FileStats>> {
  const result: Record<string, FileStats> = {};

  async function scanDirectory(path: string) {
    const files = await fs.readdir(path, { withFileTypes: true });
    for (const file of files) {
      const fullPath = join(path, file.name);
      if (file.isDirectory()) {
        await scanDirectory(fullPath);
      } else if (!file.isSymbolicLink()) {
        const ext = basename(file.name).split('.').pop()?.toLowerCase() || '(no-ext)';
        const stats = await fs.stat(fullPath);
        if (!result[ext]) {
          result[ext] = { count: 0, bytes: 0 };
        }
        result[ext].count++;
        result[ext].bytes += stats.size;
      }
    }
  }

  await scanDirectory(dirPath);
  return result;
}

export { summarizeByExtension };
