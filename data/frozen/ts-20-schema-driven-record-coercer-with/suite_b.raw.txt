test('s.string().parse returns the correct type for valid input', () => {
  expect(S.s.string().parse('hello')).toEqual({ ok: true, data: 'hello' });

test('s.string().parse returns an error for invalid input', () => {
  expect(S.s.string().parse(123)).toEqual({ ok: false, errors: [{ path: '', message: 'Expected a string, got number' }] });
});

  expect(S.s.number().parse(123)).toEqual({ ok: true, data: 123 });

});
