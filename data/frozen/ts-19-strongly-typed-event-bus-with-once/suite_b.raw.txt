- Test that `on` and `once` return idempotent unsubscribe functions.
- Test that listeners added or removed during `emit` do not disrupt or skip already-registered listeners for that same emission cycle.
- Test that events typed as `void` can be emitted with one argument (`bus.emit("reset")`), while payload events fail type-checking if the payload is omitted or wrong.
- Test that calling unsubscribe twice does not affect other listeners.
- Test that the bus can handle multiple event types with different payloads.
- Test that the bus can handle events with no payload.

```ts
import { createTypedBus } from './createTypedBus';

test('on and once return idempotent unsubscribe functions', () => {
  const bus = createTypedBus<{ click: number; reset: void }>();
  const handler = vi.fn();
  const unsubscribe = bus.on('click', handler);
  unsubscribe();
  unsubscribe(); // Should not affect other listeners
  bus.emit('click', 1);
  expect(handler).not.toHaveBeenCalled();
});

test('listeners added or removed during emit do not disrupt or skip already-registered listeners for that same emission cycle', () => {
  const bus = createTypedBus<{ click: number; reset: void }>();
  const handler1 = vi.fn();
  const handler2 = vi.fn();
  const unsubscribe1 = bus.on('click', handler1);
  bus.on('click', handler2);
  unsubscribe1();
  bus.emit('click', 1);
  expect(handler1).not.toHaveBeenCalled();
  expect(handler2).toHaveBeenCalledWith(1);
});

test('events typed as void can be emitted with one argument (bus.emit("reset"))', () => {
  const bus = createTypedBus<{ click: number; reset: void }>();
  const handler = vi.fn();
  bus.on('reset', handler);
  bus.emit('reset');
  expect(handler).toHaveBeenCalledWith(undefined);
});

test('payload events fail type-checking if the payload is omitted or wrong', () => {
  const bus = createTypedBus<{ click: number; reset: void }>();
  const handler = vi.fn();
  bus.on('click', handler);
  expect(() => bus.emit('click')).toThrow();
  expect(() => bus.emit('click', 'wrong')).toThrow();
});

test('calling unsubscribe twice does not affect other listeners', () => {
  const bus = createTypedBus<{ click: number; reset: void }>();
  const handler1 = vi.fn();
  const handler2 = vi.fn();
  const unsubscribe1 = bus.on('click', handler1);
  const unsubscribe2 = bus.on('click', handler2);
  unsubscribe1();
  unsubscribe2();
  bus.emit('click', 1);
  expect(handler1).not.toHaveBeenCalled();
  expect(handler2).not.toHaveBeenCalled();
});

test('the bus can handle multiple event types with different payloads', () => {
  const bus = createTypedBus<{ click: number; reset: void; load: string }>();
  const handler1 = vi.fn();
  const handler2 = vi.fn();
  const handler3 = vi.fn();
  bus.on('click', handler1);
  bus.on('reset', handler2);
  bus.on('load', handler3);
  bus.emit('click', 1);
  bus.emit('reset');
  bus.emit('load', 'data');
  expect(handler1).toHaveBeenCalledWith(1);
  expect(handler2).toHaveBeenCalledWith(undefined);
  expect(handler3).toHaveBeenCalledWith('data');
});

test('the bus can handle events with no payload', () => {
  const bus = createTypedBus<{ click: number; reset: void }>();
  const handler = vi.fn();
  bus.on('reset', handler);
  bus.emit('reset');
  expect(handler).toHaveBeenCalledWith(undefined);
});
```
