probe("basic request", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "pr-"));
  const configPath = path.join(dir, "config.json");
  await fs.promises.writeFile(configPath, JSON.stringify({ key: "value" }));

  const getConfig = () => JSON.parse(fs.readFileSync(configPath, "utf8"));
  const handler = createConfigHandler(getConfig);

  const req = new http.IncomingMessage({ url: "/", headers: {} });
  const res = new http.ServerResponse(req);
  handler(req, res);

  return { statusCode: res.statusCode, headers: res.getHeaders(), body: await readResponseBody(res) };
});
