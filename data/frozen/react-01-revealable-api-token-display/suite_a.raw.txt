- An empty token renders "No token provided" with no Show or Copy buttons.
- By default renders • repeated token.length times; clicking "Show" reveals the raw token and switches the button label to "Hide".
- Clicking "Copy" calls navigator.clipboard.writeText(token), displays "Copied" for 2000 ms, and clears its timer on unmount.
- Changing the token prop while revealed automatically re-masks the new token.

```tsx
import React from 'react';
import { render, screen, fireEvent, cleanup, act, waitFor } from '@testing-library/react';
import '@testing-library/jest-dom/extend-expect';
import { RevealableToken } from './RevealableToken'; // Adjust the import path as necessary

afterEach(cleanup);

test("renders 'No token provided' with no Show or Copy buttons for an empty token", () => {
  render(<RevealableToken token="" />);
  expect(screen.getByText('No token provided')).toBeInTheDocument();
  expect(screen.queryByRole('button', { name: 'Show' })).not.toBeInTheDocument();
  expect(screen.queryByRole('button', { name: 'Copy' })).not.toBeInTheDocument();
});

test("renders masked token and provides Show/Hide toggle", async () => {
  render(<RevealableToken token="12345" />);
  expect(screen.getByText('•••••')).toBeInTheDocument();
  fireEvent.click(screen.getByRole('button', { name: 'Show' }));
  await waitFor(() => expect(screen.getByText('12345')).toBeInTheDocument());
  expect(screen.getByRole('button', { name: 'Hide' })).toBeInTheDocument();
});

test("clicking 'Copy' copies the token and displays 'Copied'", async () => {
  const token = "12345";
  render(<RevealableToken token={token} />);
  fireEvent.click(screen.getByRole('button', { name: 'Show' }));
  fireEvent.click(screen.getByRole('button', { name: 'Copy' }));
  await waitFor(() => expect(screen.getByText('Copied')).toBeInTheDocument());
  expect(screen.queryByText('Copied')).not.toBeInTheDocument();
});

test("changing the token prop while revealed re-masks the new token", async () => {
  const { rerender } = render(<RevealableToken token="12345" />);
  fireEvent.click(screen.getByRole('button', { name: 'Show' }));
  rerender(<RevealableToken token="67890" />);
  await waitFor(() => expect(screen.getByText('•••••')).toBeInTheDocument());
});
```
