- Sets `isLoading` to `true` while the promise is pending.
- Captures any thrown error into `error`.
- Resets status correctly on subsequent executions.

```tsx
import { render, screen, fireEvent, waitFor, cleanup, act } from '@testing-library/react';
import { useAsyncAction } from './useAsyncAction'; // Adjust the import path as necessary

test("isLoading is true while the promise is pending", async () => {
  const mockAction = vi.fn().mockResolvedValue("Success");
  const { execute, isLoading } = useAsyncAction(mockAction);

  render(<button onClick={execute}>Click me</button>);

  fireEvent.click(screen.getByRole("button"));

  expect(isLoading).toBe(true);
});

test("captures any thrown error into error", async () => {
  const mockAction = vi.fn().mockRejectedValue(new Error("Something went wrong"));
  const { execute, error } = useAsyncAction(mockAction);

  render(<button onClick={execute}>Click me</button>);

  fireEvent.click(screen.getByRole("button"));

  await waitFor(() => expect(error).toBeInstanceOf(Error));
  expect(error?.message).toBe("Something went wrong");
});

test("resets status correctly on subsequent executions", async () => {
  const mockAction = vi.fn().mockResolvedValue("Success");
  const { execute, isLoading, error } = useAsyncAction(mockAction);

  render(<button onClick={execute}>Click me</button>);

  fireEvent.click(screen.getByRole("button"));

  await waitFor(() => expect(isLoading).toBe(false));
  expect(error).toBeNull();

  fireEvent.click(screen.getByRole("button"));

  expect(isLoading).toBe(true);
  expect(error).toBeNull();
});
```
