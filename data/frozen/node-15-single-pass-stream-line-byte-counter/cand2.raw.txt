import { Readable } from 'stream';

export async function measureStream(readable: Readable): Promise<{ bytes: number; lines: number }> {
  let totalBytes = 0;
  let totalLines = 0;
  let chunk = '';

  for await (const chunk of readable) {
    totalBytes += chunk.length;
    chunk += chunk.toString('utf-8');
    const lineBreaks = (chunk.match(/\n/g) || []).length;
    totalLines += lineBreaks;
    chunk = chunk.substring(lineBreaks);
  }

  if (chunk.length > 0) {
    totalBytes += chunk.length;
    totalLines += 1;
  }

  return { bytes: totalBytes, lines: totalLines };
}
