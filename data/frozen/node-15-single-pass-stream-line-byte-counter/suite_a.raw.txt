- An empty stream resolves to { bytes: 0, lines: 0 }.
- Counts both trailing-newline files ("a\nb\n" → 2 lines) and files without a final newline ("a\nb" → 2 lines) accurately across arbitrary chunk splits.
- Counts multi-byte UTF-8 characters by byte length rather than JavaScript string code unit length.

```ts
import { measureStream } from './measureStream'; // Adjust the import path as necessary
import { Readable } from 'stream';
import { promises as fs } from 'fs';
import { join } from 'path';
import { tmpdir } from 'os';

test("Counts bytes and lines in an empty stream", async () => {
  const result = await measureStream(Readable.from([]));
  expect(result).toEqual({ bytes: 0, lines: 0 });
});

test("Counts lines in a file with a trailing newline", async () => {
  const dir = await fs.mkdtemp(join(tmpdir(), "st-"));
  const filePath = join(dir, "file.txt");
  await fs.writeFile(filePath, "a\nb\n");
  const result = await measureStream(Readable.from([filePath]));
  expect(result).toEqual({ bytes: 5, lines: 2 });
});

test("Counts lines in a file without a trailing newline", async () => {
  const dir = await fs.mkdtemp(join(tmpdir(), "st-"));
  const filePath = join(dir, "file.txt");
  await fs.writeFile(filePath, "a\nb");
  const result = await measureStream(Readable.from([filePath]));
  expect(result).toEqual({ bytes: 4, lines: 2 });
});

test("Counts multi-byte UTF-8 characters by byte length", async () => {
  const dir = await fs.mkdtemp(join(tmpdir(), "st-"));
  const filePath = join(dir, "file.txt");
  await fs.writeFile(filePath, "你好\n世界");
  const result = await measureStream(Readable.from([filePath]));
  expect(result).toEqual({ bytes: 12, lines: 2 });
});
```
