probe("valid relative path", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "pr-"));
  const result = resolveInsideRoot(dir, "a/../b.txt");
  await fs.promises.rm(dir, { recursive: true });
  return result;
});
