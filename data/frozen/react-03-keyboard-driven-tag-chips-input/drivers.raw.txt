probe("Add tag with Enter", async () => {
  render(<TagChipsInput value={[]} onChange={vi.fn()} />);
  fireEvent.change(screen.getByRole("textbox"), { target: { value: "  tag1  " } });
  fireEvent.keyDown(screen.getByRole("textbox"), { key: "Enter" });
  return snap();
});

});
