import React, { useState, useRef } from 'react';

export function TagChipsInput({ value, onChange }: { value: string[]; onChange: (next: string[]) => void }) {
  const [inputValue, setInputValue] = useState('');
  const inputRef = useRef<HTMLInputElement>(null);

  const handleKeyDown = (event: React.KeyboardEvent<HTMLInputElement>) => {
    if (event.key === 'Enter' || event.key === ',') {
      event.preventDefault();
      const nextValue = inputValue.trim();
      if (nextValue && !value.some(v => v.toLowerCase() === nextValue.toLowerCase())) {
        onChange([...value, nextValue]);
      }
      setInputValue('');
      inputRef.current?.focus();
    }
  };

  const handleInputChange = (event: React.ChangeEvent<HTMLInputElement>) => {
    setInputValue(event.target.value);
  };

  const handleRemove = (tag: string) => {
    onChange(value.filter(v => v.toLowerCase() !== tag.toLowerCase()));
  };

  const handleBackspace = () => {
    if (inputValue === '') {
      const newValue = [...value];
      newValue.pop();
      onChange(newValue);
    }
  };

  return (
    <div>
      <input
        ref={inputRef}
        type="text"
        value={inputValue}
        onChange={handleInputChange}
        onKeyDown={handleKeyDown}
        onKeyPress={handleBackspace}
        placeholder="Add tags..."
      />
      <div>
        {value.map((tag, index) => (
          <span key={index} className="tag">
            {tag}
            <button
              aria-label={`Remove ${tag}`}
              onClick={() => handleRemove(tag)}
            >
              ✖️
            </button>
          </span>
        ))}
      </div>
      <style jsx>{`
        .tag {
          display: inline-flex;
          align-items: center;
          background-color: #f0f0f0;
          padding: 5px 10px;
          border-radius: 4px;
          margin: 2px;
        }
        .tag button {
          margin-left: 5px;
          background-color: #ff4d4f;
          color: white;
          border: none;
          border-radius: 4px;
          cursor: pointer;
        }
        .tag button:hover {
          background-color: #c23c3e;
        }
      `}</style>
    </div>
  );
}
