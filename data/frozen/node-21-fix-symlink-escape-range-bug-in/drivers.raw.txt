probe("basic file read", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "pr-"));
  const filePath = path.join(dir, "test.txt");
  await fs.promises.writeFile(filePath, "Hello, world!");
  const server = createSafeFileServer(dir);
  const response = await sendRequest(server, "GET", "/test.txt");
  await server.close();
  return response;
});
