- Resolves real filesystem paths (via realpath) and returns HTTP 403 for both path traversals (including percent-encoded) and symlinks targeting files outside rootDir.
- Returns 200 with accurate Content-Length for GET, and returns identical headers with zero body bytes for HEAD.
- Returns 404 for missing files or directories (never dumping raw directory handles or crashing the server).
- Handles empty or invalid input.
- Handles boundaries (e.g., root directory at the filesystem root).
- Handles repeated calls.
- Cleans up timers, listeners, files, or other resources.

```ts
import * as fs from "node:fs";
import * as path from "node:path";
import * as os from "node:os";
import { createSafeFileServer } from "./createSafeFileServer"; // Adjust the import path as necessary

test("returns 403 for path traversal", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const server = createSafeFileServer(dir);
  const response = await new Promise((resolve) => {
    const req = http.request({ method: "GET", url: `http://localhost:8080/../` }, resolve);
    req.end();
  });
  expect(response.statusCode).toBe(403);
  server.close();
});

test("returns 403 for symlink outside rootDir", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const outsideDir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const symlinkPath = path.join(dir, "outside");
  await fs.promises.symlink(outsideDir, symlinkPath);
  const server = createSafeFileServer(dir);
  const response = await new Promise((resolve) => {
    const req = http.request({ method: "GET", url: `http://localhost:8080/outside/` }, resolve);
    req.end();
  });
  expect(response.statusCode).toBe(403);
  server.close();
  await fs.promises.unlink(symlinkPath);
  await fs.promises.rmdir(outsideDir);
});

test("returns 200 with accurate Content-Length for GET", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const filePath = path.join(dir, "test.txt");
  await fs.promises.writeFile(filePath, "Hello, world!");
  const server = createSafeFileServer(dir);
  const response = await new Promise((resolve) => {
    const req = http.request({ method: "GET", url: `http://localhost:8080/test.txt` }, resolve);
    req.end();
  });
  expect(response.statusCode).toBe(200);
  expect(response.headers["content-length"]).toBe("13");
  server.close();
});

test("returns identical headers with zero body bytes for HEAD", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const filePath = path.join(dir, "test.txt");
  await fs.promises.writeFile(filePath, "Hello, world!");
  const server = createSafeFileServer(dir);
  const response = await new Promise((resolve) => {
    const req = http.request({ method: "HEAD", url: `http://localhost:8080/test.txt` }, resolve);
    req.end();
  });
  expect(response.statusCode).toBe(200);
  expect(response.headers["content-length"]).toBe("13");
  expect(response.headers["content-type"]).toBe("text/plain");
  server.close();
});

test("returns 404 for missing files or directories", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const server = createSafeFileServer(dir);
  const response = await new Promise((resolve) => {
    const req = http.request({ method: "GET", url: `http://localhost:8080/nonexistent.txt` }, resolve);
    req.end();
  });
  expect(response.statusCode).toBe(404);
  server.close();
});
```
