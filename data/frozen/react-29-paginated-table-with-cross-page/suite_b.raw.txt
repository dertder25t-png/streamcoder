- Test that the table renders with the correct number of rows based on `pageSize`.
- Test that the header checkbox toggles the selection of all visible rows on the current page.
- Test that selections made on page 1 persist when navigating to page 2 and back.
- Test that the "Clear all (N)" button clears all selections and updates the count.
- Test that the table handles an empty `rows` array gracefully.
- Test that the table handles a `pageSize` of 0 gracefully.
- Test that the table clamps the active page to the last valid page when the current page exceeds the new total page count.

```tsx
import React from 'react';
import { render, screen, fireEvent, cleanup, act, waitFor } from '@testing-library/react';
import '@testing-library/jest-dom/extend-expect';
import PaginatedSelectorTable from './PaginatedSelectorTable';

afterEach(cleanup);

test("renders with the correct number of rows based on pageSize", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },
    { id: '5', name: 'Row 5' },
  ];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  await waitFor(() => {
    expect(screen.getAllByRole('checkbox')).toHaveLength(pageSize);
  });
});

test("header checkbox toggles the selection of all visible rows on the current page", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },
    { id: '5', name: 'Row 5' },
  ];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  await waitFor(() => {
    fireEvent.click(screen.getByRole('checkbox', { name: 'Select page' }));
    expect(onSelectionChange).toHaveBeenCalledWith(['1', '2']);
  });
});

test("selections made on page 1 persist when navigating to page 2 and back", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },
    { id: '5', name: 'Row 5' },
  ];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  await waitFor(() => {
    fireEvent.click(screen.getByRole('checkbox', { name: 'Select Row 1' }));
    fireEvent.click(screen.getByRole('button', { name: /next/i }));
    fireEvent.click(screen.getByRole('button', { name: /prev/i }));
    expect(onSelectionChange).toHaveBeenCalledWith(['1']);
  });
});

test("Clear all (N) button clears all selections and updates the count", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },
    { id: '5', name: 'Row 5' },
  ];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  await waitFor(() => {
    fireEvent.click(screen.getByRole('checkbox', { name: 'Select Row 1' }));
    fireEvent.click(screen.getByRole('checkbox', { name: 'Select Row 2' }));
    fireEvent.click(screen.getByRole('button', { name: 'Clear all (2)' }));
    expect(onSelectionChange).toHaveBeenCalledWith([]);
  });
});

test("handles an empty rows array gracefully", async () => {
  const rows: { id: string; name: string }[] = [];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  await waitFor(() => {
    expect(screen.getAllByRole('checkbox')).toHaveLength(0);
  });
});

test("handles a pageSize of 0 gracefully", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },

test("clamps the active page to the last valid page when the current page exceeds the new total page count", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },
    { id: '5', name: 'Row 5' },
  ];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  await waitFor(() => {

    fireEvent.click(screen.getByRole('button', { name: /next/i }));

    expect(screen.getByRole('button', { name: /prev/i })).toBeDisabled();
  });
```
