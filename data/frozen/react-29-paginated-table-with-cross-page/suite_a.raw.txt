- Selections made on page 1 persist when navigating to page 2 and back, and onSelectionChange reports all selected IDs across every page.
- The header checkbox reflects and toggles only the rows visible on the current page without deselecting rows on other pages.
- If the rows prop shrinks so the current page exceeds the new total page count, the active page clamps automatically to the last valid page.

```tsx
import React from 'react';
import { render, screen, fireEvent, cleanup, act, waitFor } from '@testing-library/react';
import PaginatedSelectorTable from './PaginatedSelectorTable';

afterEach(cleanup);

test("Selections made on page 1 persist when navigating to page 2 and back, and onSelectionChange reports all selected IDs across every page", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },
    { id: '5', name: 'Row 5' },
  ];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  // Select Row 1 on page 1
  fireEvent.click(screen.getByRole('checkbox', { name: 'Select Row 1' }));
  expect(onSelectionChange).toHaveBeenCalledWith(['1']);

  // Navigate to page 2
  fireEvent.click(screen.getByRole('button', { name: /next/i }));
  await waitFor(() => expect(screen.getByRole('checkbox', { name: 'Select Row 3' })).toBeInTheDocument());

  // Select Row 3 on page 2
  fireEvent.click(screen.getByRole('checkbox', { name: 'Select Row 3' }));
  expect(onSelectionChange).toHaveBeenCalledWith(['1', '3']);

  // Navigate back to page 1
  fireEvent.click(screen.getByRole('button', { name: /prev/i }));
  await waitFor(() => expect(screen.getByRole('checkbox', { name: 'Select Row 1' })).toBeInTheDocument());

  // Verify Row 1 is still selected on page 1
  expect(onSelectionChange).toHaveBeenCalledWith(['1', '3']);
});

test("The header checkbox reflects and toggles only the rows visible on the current page without deselecting rows on other pages", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },
    { id: '5', name: 'Row 5' },
  ];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  // Select Row 1 on page 1
  fireEvent.click(screen.getByRole('checkbox', { name: 'Select Row 1' }));
  expect(onSelectionChange).toHaveBeenCalledWith(['1']);

  // Toggle header checkbox on page 1
  fireEvent.click(screen.getByRole('checkbox', { name: 'Select page' }));
  expect(onSelectionChange).toHaveBeenCalledWith(['1', '2']);

  // Navigate to page 2
  fireEvent.click(screen.getByRole('button', { name: /next/i }));
  await waitFor(() => expect(screen.getByRole('checkbox', { name: 'Select Row 3' })).toBeInTheDocument());

  // Toggle header checkbox on page 2
  fireEvent.click(screen.getByRole('checkbox', { name: 'Select page' }));
  expect(onSelectionChange).toHaveBeenCalledWith(['1', '2', '3', '4']);

  // Navigate back to page 1
  fireEvent.click(screen.getByRole('button', { name: /prev/i }));
  await waitFor(() => expect(screen.getByRole('checkbox', { name: 'Select Row 1' })).toBeInTheDocument());

  // Verify Row 1 is still selected on page 1
  expect(onSelectionChange).toHaveBeenCalledWith(['1', '2', '3', '4']);
});

test("If the rows prop shrinks so the current page exceeds the new total page count, the active page clamps automatically to the last valid page", async () => {
  const rows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
    { id: '4', name: 'Row 4' },
    { id: '5', name: 'Row 5' },
  ];
  const pageSize = 2;
  const onSelectionChange = vi.fn();

  render(<PaginatedSelectorTable rows={rows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);

  // Navigate to page 2
  fireEvent.click(screen.getByRole('button', { name: /next/i }));
  await waitFor(() => expect(screen.getByRole('checkbox', { name: 'Select Row 3' })).toBeInTheDocument());

  // Shrink rows to 3
  const newRows = [
    { id: '1', name: 'Row 1' },
    { id: '2', name: 'Row 2' },
    { id: '3', name: 'Row 3' },
  ];

  // Update rows prop
  act(() => {
    render(<PaginatedSelectorTable rows={newRows} pageSize={pageSize} onSelectionChange={onSelectionChange} />);
  });

  // Verify active page clamps to last valid page (page 1)
  expect(screen.getByRole('button', { name: /prev/i })).toBeDisabled();
  expect(screen.getByRole('button', { name: /next/i })).toBeEnabled();
});
```
