probe("file exists and is valid JSON", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "pr-"));
  const filePath = path.join(dir, "test.json");
  await fs.promises.writeFile(filePath, '{"key": "value"}');
  const result = await readJsonSafe<{ key: string }>(filePath, { key: "default" });
  return result;
});
