```typescript
import { Transaction, Bucket, groupTransactionsByCurrency } from './path-to-module';

export interface Transaction {
  id: string;
  currency: string;
  amountMinor: number; // integer minor units, e.g. cents
}

export interface Bucket {
  count: number;
  totalMinor: number;
  ids: string[];
}

const DEFAULT_BUCKET: Bucket = { count: 0, totalMinor: 0, ids: [] };
const cache: Record<string, Bucket> = {};

export function groupTransactionsByCurrency(items: Transaction[]): Record<string, Bucket> {
  for (const item of items) {
    const bucket = cache[item.currency] ?? { ...DEFAULT_BUCKET };
    bucket.count++;
    bucket.totalMinor += item.amountMinor;
    bucket.ids.push(item.id);
    cache[item.currency] = bucket;
  }
  return { ...cache };
}
```
