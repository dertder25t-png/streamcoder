- Clicking Undo before graceMs cancels the timer and prevents onCommitDelete from being called.
- Staggered deletions run independent timers so undoing one item does not alter another item's deadline.
- If onCommitDelete changes identity before a timer fires, the latest callback reference is invoked.
- External removal of an item from items or component unmount cancels pending timers so stale deletions never fire.

```tsx
import React from 'react';
import { render, screen, fireEvent, cleanup, act, waitFor } from '@testing-library/react';
import { DeferredActionList } from './DeferredActionList';

afterEach(cleanup);

test('Undo cancels the timer and prevents onCommitDelete from being called', async () => {
  const onCommitDelete = vi.fn();
  render(<DeferredActionList items={[{ id: '1', label: 'Alpha' }]} onCommitDelete={onCommitDelete} />);
  const deleteButton = screen.getByRole('button', { name: 'Delete Alpha' });
  fireEvent.click(deleteButton);
  await waitFor(() => screen.getByText('Deleting...'));
  const undoButton = screen.getByRole('button', { name: 'Undo' });
  fireEvent.click(undoButton);
  expect(onCommitDelete).not.toHaveBeenCalled();
});

test('Staggered deletions run independent timers', async () => {
  vi.useFakeTimers();
  const onCommitDelete = vi.fn();
  render(<DeferredActionList items={[{ id: '1', label: 'Alpha' }, { id: '2', label: 'Beta' }]} onCommitDelete={onCommitDelete} />);
  const deleteButton1 = screen.getByRole('button', { name: 'Delete Alpha' });
  fireEvent.click(deleteButton1);
  const deleteButton2 = screen.getByRole('button', { name: 'Delete Beta' });
  fireEvent.click(deleteButton2);
  await waitFor(() => screen.getByText('Deleting...'));
  vi.advanceTimersByTime(2000);
  expect(onCommitDelete).not.toHaveBeenCalled();
  fireEvent.click(screen.getByRole('button', { name: 'Undo' }));
  vi.advanceTimersByTime(2000);
  expect(onCommitDelete).toHaveBeenCalledWith('2');
  expect(onCommitDelete).not.toHaveBeenCalledWith('1');
  vi.useRealTimers();
});

test('If onCommitDelete changes identity before a timer fires, the latest callback reference is invoked', async () => {
  const onCommitDelete1 = vi.fn();
  const onCommitDelete2 = vi.fn();
  render(<DeferredActionList items={[{ id: '1', label: 'Alpha' }]} onCommitDelete={onCommitDelete1} />);
  const deleteButton = screen.getByRole('button', { name: 'Delete Alpha' });
  fireEvent.click(deleteButton);
  await waitFor(() => screen.getByText('Deleting...'));
  render(<DeferredActionList items={[{ id: '1', label: 'Alpha' }]} onCommitDelete={onCommitDelete2} />);
  vi.advanceTimersByTime(4000);
  expect(onCommitDelete2).toHaveBeenCalledWith('1');
  expect(onCommitDelete1).not.toHaveBeenCalled();
});

test('External removal of an item from items or component unmount cancels pending timers', async () => {
  vi.useFakeTimers();
  const onCommitDelete = vi.fn();
  const { rerender } = render(<DeferredActionList items={[{ id: '1', label: 'Alpha' }]} onCommitDelete={onCommitDelete} />);
  const deleteButton = screen.getByRole('button', { name: 'Delete Alpha' });
  fireEvent.click(deleteButton);
  await waitFor(() => screen.getByText('Deleting...'));
  rerender(<DeferredActionList items={[]} onCommitDelete={onCommitDelete} />);
  vi.advanceTimersByTime(4000);
  expect(onCommitDelete).not.toHaveBeenCalled();
  vi.useRealTimers();
});
```
