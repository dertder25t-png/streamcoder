probe("basic deletion", async () => {
  vi.useFakeTimers();
  const { unmount } = render(<DeferredActionList items={[{ id: "1", label: "Alpha" }]} onCommitDelete={vi.fn()} />);
  fireEvent.click(screen.getByRole("button", { name: "Delete Alpha" }));
  expect(screen.getByText("Deleting...")).toBeInTheDocument();
  expect(screen.getByRole("button", { name: "Undo" })).toBeInTheDocument();
  vi.advanceTimersByTime(3000);
  expect(screen.getByText("Alpha")).not.toBeInTheDocument();
  unmount();
  return snap();
});

  expect(screen.getByRole("button", { name: "Undo Beta" })).toBeInTheDocument();
  vi.advanceTimersByTime(3000);
  expect(screen.getByText("Alpha")).not.toBeInTheDocument();
  expect(screen.getByText("Beta")).not.toBeInTheDocument();
  unmount();
  return snap();
});
