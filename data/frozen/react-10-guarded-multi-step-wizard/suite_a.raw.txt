- Back is disabled on step 1
- Next is disabled whenever the current step has isValid === false
- On the last step, the primary button displays "Finish" and invokes onComplete() only when isValid !== false
- Moving backward and forward never lets the active index drop below 0 or exceed steps.length - 1

```tsx
import React from 'react';
import { render, screen, fireEvent, cleanup, act, waitFor } from '@testing-library/react';
import StepWizard from './StepWizard';

afterEach(cleanup);

test("Back is disabled on step 1", async () => {
  const steps = [
    { id: '1', title: 'Step 1', content: 'Content 1' },
    { id: '2', title: 'Step 2', content: 'Content 2' }
  ];
  const onComplete = vi.fn();

  render(<StepWizard steps={steps} onComplete={onComplete} />);
  await waitFor(() => expect(screen.getByRole('button', { name: 'Back' })).toBeDisabled());
});

test("Next is disabled whenever the current step has isValid === false", async () => {
  const steps = [
    { id: '1', title: 'Step 1', content: 'Content 1', isValid: false },
    { id: '2', title: 'Step 2', content: 'Content 2' }
  ];
  const onComplete = vi.fn();

  render(<StepWizard steps={steps} onComplete={onComplete} />);
  await waitFor(() => expect(screen.getByRole('button', { name: 'Next' })).toBeDisabled());
});

test("On the last step, the primary button displays 'Finish' and invokes onComplete() only when isValid !== false", async () => {
  const steps = [
    { id: '1', title: 'Step 1', content: 'Content 1' },
    { id: '2', title: 'Step 2', content: 'Content 2', isValid: true }
  ];
  const onComplete = vi.fn();

  render(<StepWizard steps={steps} onComplete={onComplete} />);
  await waitFor(() => expect(screen.getByRole('button', { name: 'Finish' })).toBeEnabled());
  fireEvent.click(screen.getByRole('button', { name: 'Finish' }));
  await waitFor(() => expect(onComplete).toHaveBeenCalled());
});

test("Moving backward and forward never lets the active index drop below 0 or exceed steps.length - 1", async () => {
  const steps = [
    { id: '1', title: 'Step 1', content: 'Content 1' },
    { id: '2', title: 'Step 2', content: 'Content 2' }
  ];
  const onComplete = vi.fn();

  render(<StepWizard steps={steps} onComplete={onComplete} />);
  await waitFor(() => expect(screen.getByRole('button', { name: 'Back' })).toBeDisabled());
  fireEvent.click(screen.getByRole('button', { name: 'Next' }));
  await waitFor(() => expect(screen.getByRole('button', { name: 'Back' })).toBeEnabled());
  fireEvent.click(screen.getByRole('button', { name: 'Back' }));
  await waitFor(() => expect(screen.getByRole('button', { name: 'Back' })).toBeDisabled());
});
```
