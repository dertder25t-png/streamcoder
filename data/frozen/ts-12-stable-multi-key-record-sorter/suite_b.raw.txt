- Returns a new sorted array without mutating items.
- Secondary keys break ties when primary keys are equal, honoring independent "asc" and "desc" directions.
- Records equal across all specified keys preserve their original relative order.
- Handles empty input.
- Handles invalid input.
- Handles repeated calls.
- Handles cleanup of resources.

```ts
import { sortRecordsByKeys } from './sortRecordsByKeys';

test("sorts records by primary key in ascending order", () => {
  const items = [{ name: "Alice", age: 25 }, { name: "Bob", age: 30 }];
  const specs = [{ key: "age" }];
  const sortedItems = sortRecordsByKeys(items, specs);
  expect(sortedItems).toEqual([{ name: "Alice", age: 25 }, { name: "Bob", age: 30 }]);
});

test("sorts records by primary key in descending order", () => {
  const items = [{ name: "Alice", age: 25 }, { name: "Bob", age: 30 }];
  const specs = [{ key: "age", direction: "desc" }];
  const sortedItems = sortRecordsByKeys(items, specs);
  expect(sortedItems).toEqual([{ name: "Bob", age: 30 }, { name: "Alice", age: 25 }]);
});

test("breaks ties with secondary keys in ascending order", () => {
  const items = [{ name: "Alice", age: 25, score: 80 }, { name: "Bob", age: 25, score: 90 }];
  const specs = [{ key: "age" }, { key: "score" }];
  const sortedItems = sortRecordsByKeys(items, specs);
  expect(sortedItems).toEqual([{ name: "Alice", age: 25, score: 80 }, { name: "Bob", age: 25, score: 90 }]);
});

test("breaks ties with secondary keys in descending order", () => {
  const items = [{ name: "Alice", age: 25, score: 80 }, { name: "Bob", age: 25, score: 90 }];
  const specs = [{ key: "age" }, { key: "score", direction: "desc" }];
  const sortedItems = sortRecordsByKeys(items, specs);
  expect(sortedItems).toEqual([{ name: "Bob", age: 25, score: 90 }, { name: "Alice", age: 25, score: 80 }]);
});

test("preserves original order for records equal across all specified keys", () => {
  const items = [{ name: "Alice", age: 25, score: 80 }, { name: "Bob", age: 25, score: 80 }];
  const specs = [{ key: "score" }];
  const sortedItems = sortRecordsByKeys(items, specs);
  expect(sortedItems).toEqual([{ name: "Alice", age: 25, score: 80 }, { name: "Bob", age: 25, score: 80 }]);
});

test("handles empty input", () => {
  const items: any[] = [];
  const specs = [{ key: "age" }];
  const sortedItems = sortRecordsByKeys(items, specs);
  expect(sortedItems).toEqual([]);
});

test("handles invalid input", () => {
  const items = [{ name: "Alice", age: 25 }];
  const specs = [{ key: "invalidKey" }];
  const sortedItems = sortRecordsByKeys(items, specs);
  expect(sortedItems).toEqual([{ name: "Alice", age: 25 }]);
});

test("handles repeated calls", () => {
  const items = [{ name: "Alice", age: 25 }, { name: "Bob", age: 30 }];
  const specs = [{ key: "age" }];
  const sortedItems1 = sortRecordsByKeys(items, specs);
  const sortedItems2 = sortRecordsByKeys(items, specs);
  expect(sortedItems1).toEqual([{ name: "Alice", age: 25 }, { name: "Bob", age: 30 }]);
  expect(sortedItems2).toEqual([{ name: "Alice", age: 25 }, { name: "Bob", age: 30 }]);
});
```
