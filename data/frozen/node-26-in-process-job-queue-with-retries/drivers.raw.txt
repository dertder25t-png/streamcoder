  const jobQueue = createJobQueue({ concurrency: 2, maxRetries: 3, processor: async () => {}, onDeadLetter: () => {} });
  jobQueue.enqueue({ id: 1 });
  jobQueue.enqueue({ id: 2 });
  jobQueue.enqueue({ id: 3 });
  return jobQueue;
});

probe("drain job queue", async () => {
  const jobQueue = createJobQueue({ concurrency: 2, maxRetries: 3, processor: async () => {}, onDeadLetter: () => {} });
  jobQueue.enqueue({ id: 1 });
  jobQueue.enqueue({ id: 2 });
  jobQueue.enqueue({ id: 3 });
  await jobQueue.drain();
  return jobQueue;
});

  const jobQueue = createJobQueue({ concurrency: 2, maxRetries: 3, processor: async () => {}, onDeadLetter: () => {} });
  jobQueue.enqueue({ id: 1 });
  jobQueue.enqueue({ id: 2 });
  jobQueue.enqueue({ id: 3 });
  const stats = jobQueue.stats();
  return stats;
});
