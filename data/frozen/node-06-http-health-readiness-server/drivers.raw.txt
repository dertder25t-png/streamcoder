probe("GET /live returns 200", async () => {
  const server = createHealthServer(() => Promise.resolve(true));
  const response = await vi.fn(() => ({ status: 200, headers: { "Content-Type": "application/json" }, body: JSON.stringify({ status: "ok" }) }));
  server.on("request", response);
  server.listen(0, () => {
    const port = (server.address() as { port: number }).port;
    return fetch(`http://localhost:${port}/live`).then(res => res.json());
  });
});
