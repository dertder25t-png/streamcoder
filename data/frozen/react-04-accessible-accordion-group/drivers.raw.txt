probe("Single item, toggle open and close", async () => {
  render(<AccordionGroup items={[{ id: "1", title: "Item 1", content: "Content 1" }]} />);
  fireEvent.click(screen.getByRole("button", { name: "Item 1" }));
  await waitFor(() => expect(screen.getByRole("button", { name: "Item 1" })).toHaveAttribute("aria-expanded", "true"));
  fireEvent.click(screen.getByRole("button", { name: "Item 1" }));
  await waitFor(() => expect(screen.getByRole("button", { name: "Item 1" })).toHaveAttribute("aria-expanded", "false"));
  return snap();
});
