- When `allowMultiple` is false, opening one item automatically closes any previously open item.
- When `allowMultiple` is false, clicking the open header closes it.
- When `allowMultiple` is true, multiple panels can remain open simultaneously.
- Each trigger button sets accurate `aria-expanded` and links to its panel via `aria-controls`.

```tsx
import React from 'react';
import { render, screen, fireEvent, cleanup, act, waitFor } from '@testing-library/react';
import AccordionGroup from './AccordionGroup';

afterEach(cleanup);

test("When allowMultiple is false, opening one item automatically closes any previously open item", async () => {
  const items = [
    { id: '1', title: 'Item 1', content: 'Content 1' },
    { id: '2', title: 'Item 2', content: 'Content 2' },
  ];

  render(<AccordionGroup items={items} allowMultiple={false} />);

  const header1 = screen.getByRole('button', { name: 'Item 1' });
  const header2 = screen.getByRole('button', { name: 'Item 2' });

  fireEvent.click(header1);
  await waitFor(() => expect(header1).toHaveAttribute('aria-expanded', 'true'));
  expect(header2).toHaveAttribute('aria-expanded', 'false');

  fireEvent.click(header2);
  await waitFor(() => expect(header2).toHaveAttribute('aria-expanded', 'true'));
  expect(header1).toHaveAttribute('aria-expanded', 'false');
});

test("When allowMultiple is false, clicking the open header closes it", async () => {
  const items = [
    { id: '1', title: 'Item 1', content: 'Content 1' },
  ];

  render(<AccordionGroup items={items} allowMultiple={false} />);

  const header1 = screen.getByRole('button', { name: 'Item 1' });

  fireEvent.click(header1);
  await waitFor(() => expect(header1).toHaveAttribute('aria-expanded', 'true'));

  fireEvent.click(header1);
  await waitFor(() => expect(header1).toHaveAttribute('aria-expanded', 'false'));
});

test("When allowMultiple is true, multiple panels can remain open simultaneously", async () => {
  const items = [
    { id: '1', title: 'Item 1', content: 'Content 1' },
    { id: '2', title: 'Item 2', content: 'Content 2' },
  ];

  render(<AccordionGroup items={items} allowMultiple={true} />);

  const header1 = screen.getByRole('button', { name: 'Item 1' });
  const header2 = screen.getByRole('button', { name: 'Item 2' });

  fireEvent.click(header1);
  await waitFor(() => expect(header1).toHaveAttribute('aria-expanded', 'true'));
  expect(header2).toHaveAttribute('aria-expanded', 'false');

  fireEvent.click(header2);
  await waitFor(() => expect(header2).toHaveAttribute('aria-expanded', 'true'));
  expect(header1).toHaveAttribute('aria-expanded', 'true');
});

test("Each trigger button sets accurate aria-expanded and links to its panel via aria-controls", async () => {
  const items = [
    { id: '1', title: 'Item 1', content: 'Content 1' },
  ];

  render(<AccordionGroup items={items} allowMultiple={false} />);

  const header1 = screen.getByRole('button', { name: 'Item 1' });
  const panel1 = screen.getByRole('region', { name: 'Content 1' });

  expect(header1).toHaveAttribute('aria-expanded', 'false');
  expect(header1).toHaveAttribute('aria-controls', 'panel-1');

  fireEvent.click(header1);
  await waitFor(() => expect(header1).toHaveAttribute('aria-expanded', 'true'));
  expect(panel1).toBeVisible();
});
```
