probe("basic toast dismisses after duration", async () => {
  vi.useFakeTimers();
  render(<ToastList toasts={[{ id: "1", message: "Hello" }]} onDismiss={vi.fn()} />);
  await waitFor(() => screen.getByText("Hello"));
  vi.advanceTimersByTime(4000);
  return snap();
});

probe("manual dismiss", async () => {
  render(<ToastList toasts={[{ id: "1", message: "Hello" }]} onDismiss={vi.fn()} />);
  await waitFor(() => screen.getByText("Hello"));
  fireEvent.click(screen.getByText("Hello"));
  return snap();
});

probe("pause on hover", async () => {
  vi.useFakeTimers();
  render(<ToastList toasts={[{ id: "1", message: "Hello" }]} onDismiss={vi.fn()} />);
  await waitFor(() => screen.getByText("Hello"));
  fireEvent.mouseEnter(screen.getByText("Hello"));
  vi.advanceTimersByTime(2000);
  fireEvent.mouseLeave(screen.getByText("Hello"));
  vi.advanceTimersByTime(2000);
  return snap();
});

probe("no auto-dismiss for duration 0", async () => {
  vi.useFakeTimers();
  render(<ToastList toasts={[{ id: "1", message: "Hello", durationMs: 0 }]} onDismiss={vi.fn()} />);
  await waitFor(() => screen.getByText("Hello"));
  vi.advanceTimersByTime(4000);
  return snap();
});
