probe("basic debouncing", async () => {
  const callback = vi.fn();
  const debouncedCallback = useDebouncedCallback(callback, 100);

  fireEvent.input(screen.getByRole("textbox"), { target: { value: "a" } });
  fireEvent.input(screen.getByRole("textbox"), { target: { value: "ab" } });
  fireEvent.input(screen.getByRole("textbox"), { target: { value: "abc" } });

  vi.advanceTimersByTime(100);
  return snap();
});

probe("callback change", async () => {
  const callback1 = vi.fn();
  const callback2 = vi.fn();
  const debouncedCallback = useDebouncedCallback(callback1, 100);

  fireEvent.input(screen.getByRole("textbox"), { target: { value: "a" } });
  debouncedCallback = useDebouncedCallback(callback2, 100);
  fireEvent.input(screen.getByRole("textbox"), { target: { value: "b" } });

  vi.advanceTimersByTime(100);
  return snap();
});

probe("unmount clears timer", async () => {
  const callback = vi.fn();
  const debouncedCallback = useDebouncedCallback(callback, 100);

  fireEvent.input(screen.getByRole("textbox"), { target: { value: "a" } });
  cleanup();

  vi.advanceTimersByTime(100);
  return snap();
});
