- Returns app-2026-05-01.1.log when no logs exist for that date.
- Finds the numeric maximum rather than lexicographic maximum (so after .9.log and .10.log, it returns .11.log).
- Ignores unrelated files and logs from other dates in the same directory.
- Handles empty or invalid input gracefully.
- Handles repeated calls correctly.
- Cleans up files and resources after each test.

```ts
import { nextLogFilePath } from './nextLogFilePath'; // Adjust the import path as necessary
import fs from 'fs/promises';
import path from 'path';
import os from 'os';

afterEach(async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  await fs.promises.rmdir(dir, { recursive: true });
});

test("Returns app-2026-05-01.1.log when no logs exist for that date", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const result = await nextLogFilePath(dir, "2026-05-01");
  expect(result).toBe(path.join(dir, "app-2026-05-01.1.log"));
});

test("Finds the numeric maximum rather than lexicographic maximum", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  await fs.promises.writeFile(path.join(dir, "app-2026-05-01.9.log"), "");
  await fs.promises.writeFile(path.join(dir, "app-2026-05-01.10.log"), "");
  const result = await nextLogFilePath(dir, "2026-05-01");
  expect(result).toBe(path.join(dir, "app-2026-05-01.11.log"));
});

test("Ignores unrelated files and logs from other dates in the same directory", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  await fs.promises.writeFile(path.join(dir, "app-2026-05-01.1.log"), "");
  await fs.promises.writeFile(path.join(dir, "app-2026-05-02.1.log"), "");
  const result = await nextLogFilePath(dir, "2026-05-01");
  expect(result).toBe(path.join(dir, "app-2026-05-01.2.log"));
});

test("Handles empty or invalid input gracefully", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  await expect(nextLogFilePath(dir, "")).rejects.toThrow();
  await expect(nextLogFilePath(dir, "2026-05-01a")).rejects.toThrow();
  await expect(nextLogFilePath(dir, "2026-05-01-123")).rejects.toThrow();
});

test("Handles repeated calls correctly", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  await fs.promises.writeFile(path.join(dir, "app-2026-05-01.1.log"), "");
  const result1 = await nextLogFilePath(dir, "2026-05-01");
  const result2 = await nextLogFilePath(dir, "2026-05-01");
  expect(result1).toBe(path.join(dir, "app-2026-05-01.2.log"));
  expect(result2).toBe(path.join(dir, "app-2026-05-01.3.log"));
});
```
