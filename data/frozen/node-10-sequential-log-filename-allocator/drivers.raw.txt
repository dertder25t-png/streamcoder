probe("no logs exist", async () => {

  const result = await nextLogFilePath(dir, "2026-05-01");

probe("one log exists", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "pr-"));
  await fs.promises.writeFile(path.join(dir, "app-2026-05-01.1.log"), "");
  const result = await nextLogFilePath(dir, "2026-05-01");
  await fs.promises.unlink(path.join(dir, "app-2026-05-01.1.log"));
  await fs.promises.rmdir(dir, { recursive: true });
  return result;
});
