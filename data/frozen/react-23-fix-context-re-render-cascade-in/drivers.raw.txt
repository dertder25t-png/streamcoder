probe("increment unread count", async () => {
  const { getByRole, getByText, rerender } = render(
    <SplitStoreProvider initialUnreadCount={0}>
      <div>Unread Count: {useUnreadCount()}</div>
      <button onClick={() => useStoreActions().incrementUnread()}>Increment</button>
    </SplitStoreProvider>
  );

  fireEvent.click(getByText("Increment"));
  await waitFor(() => expect(getByText("Unread Count: 1")).toBeInTheDocument());
  rerender(
    <SplitStoreProvider initialUnreadCount={0}>
      <div>Unread Count: {useUnreadCount()}</div>
      <button onClick={() => useStoreActions().incrementUnread()}>Increment</button>
    </SplitStoreProvider>
  );
  return snap();
});
