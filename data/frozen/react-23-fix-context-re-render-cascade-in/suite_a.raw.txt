- Incrementing unreadCount re-renders only components subscribed to unreadCount, leaving useUserName and useStoreActions consumers with zero additional renders.
- Updating userName similarly does not re-render useUnreadCount consumers.
- All hooks throw a clear Error when invoked outside SplitStoreProvider.

```tsx
import React from 'react';
import { render, screen, fireEvent, cleanup, act, waitFor } from '@testing-library/react';
import SplitStoreProvider, { useUserName, useUnreadCount, useStoreActions } from './SplitStoreProvider';

afterEach(cleanup);

test("useUnreadCount re-renders when unreadCount is incremented", async () => {
  const { getByRole } = render(
    <SplitStoreProvider initialUnreadCount={0}>
      <div data-testid="unread-count">{useUnreadCount()}</div>
      <button onClick={() => useStoreActions().incrementUnread()}>Increment</button>
    </SplitStoreProvider>
  );

  expect(getByRole('button')).toBeInTheDocument();
  expect(getByRole('button')).toHaveTextContent('Increment');

  fireEvent.click(getByRole('button'));

  await waitFor(() => expect(getByRole('button')).toHaveTextContent('Increment'));
  expect(getByRole('button')).toBeInTheDocument();
  expect(getByRole('button')).toHaveTextContent('Increment');
});

test("useUserName does not re-render when unreadCount is incremented", async () => {
  const { getByRole } = render(
    <SplitStoreProvider initialUserName="John" initialUnreadCount={0}>
      <div data-testid="user-name">{useUserName()}</div>
      <button onClick={() => useStoreActions().incrementUnread()}>Increment</button>
    </SplitStoreProvider>
  );

  expect(getByRole('button')).toBeInTheDocument();
  expect(getByRole('button')).toHaveTextContent('Increment');

  fireEvent.click(getByRole('button'));

  await waitFor(() => expect(getByRole('button')).toHaveTextContent('Increment'));
  expect(getByRole('button')).toBeInTheDocument();
  expect(getByRole('button')).toHaveTextContent('Increment');
});

test("useStoreActions does not re-render when unreadCount is incremented", async () => {
  const { getByRole } = render(
    <SplitStoreProvider initialUserName="John" initialUnreadCount={0}>
      <button onClick={() => useStoreActions().incrementUnread()}>Increment</button>
    </SplitStoreProvider>
  );

  expect(getByRole('button')).toBeInTheDocument();
  expect(getByRole('button')).toHaveTextContent('Increment');

  fireEvent.click(getByRole('button'));

  await waitFor(() => expect(getByRole('button')).toHaveTextContent('Increment'));
  expect(getByRole('button')).toBeInTheDocument();
  expect(getByRole('button')).toHaveTextContent('Increment');
});

test("useUserName throws an error when invoked outside SplitStoreProvider", () => {
  expect(() => useUserName()).toThrowError("Store hooks must be used inside <SplitStoreProvider>");
});

test("useUnreadCount throws an error when invoked outside SplitStoreProvider", () => {
  expect(() => useUnreadCount()).toThrowError("Store hooks must be used inside <SplitStoreProvider>");
});

test("useStoreActions throws an error when invoked outside SplitStoreProvider", () => {
  expect(() => useStoreActions()).toThrowError("Store hooks must be used inside <SplitStoreProvider>");
});
```
