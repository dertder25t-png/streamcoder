- An empty input array returns [].
- Consecutive identical statuses collapse into one run with start equal to the first timestamp, end equal to the last timestamp, and count equal to the merged length.
- Non-consecutive repeats (e.g., "up", "down", "up") remain separate runs in order.
- The input array is never mutated and the return type preserves a narrow string literal union rather than widening to string.

```ts
import { collapseStatusLog } from './collapseStatusLog';

test("An empty input array returns []", () => {
  expect(collapseStatusLog([])).toEqual([]);
});

test("Consecutive identical statuses collapse into one run", () => {
  expect(collapseStatusLog([
    { timestamp: 1, status: "up" },
    { timestamp: 2, status: "up" },
    { timestamp: 3, status: "up" }
  ])).toEqual([
    { status: "up", start: 1, end: 3, count: 3 }
  ]);
});

test("Non-consecutive repeats remain separate runs", () => {
  expect(collapseStatusLog([
    { timestamp: 1, status: "up" },
    { timestamp: 2, status: "down" },
    { timestamp: 3, status: "up" }
  ])).toEqual([
    { status: "up", start: 1, end: 1, count: 1 },
    { status: "down", start: 2, end: 2, count: 1 },
    { status: "up", start: 3, end: 3, count: 1 }
  ]);
});

test("The input array is never mutated", () => {
  const input = [
    { timestamp: 1, status: "up" },
    { timestamp: 2, status: "up" },
    { timestamp: 3, status: "up" }
  ];
  collapseStatusLog(input);
  expect(input).toEqual([
    { timestamp: 1, status: "up" },
    { timestamp: 2, status: "up" },
    { timestamp: 3, status: "up" }
  ]);
});
```
