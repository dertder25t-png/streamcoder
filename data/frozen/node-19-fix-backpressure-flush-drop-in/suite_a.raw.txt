- Prefixes every logical line exactly once—even when lines span multiple small chunks or arrive many per chunk.
- Flushes the final line cleanly when the input ends without a trailing newline, and emits nothing for a 0-byte input.
- Cooperates cleanly with stream.pipeline into a slow writable stream with a tiny highWaterMark without unbounded memory growth or data loss.

```ts
import { createLinePrefixStream } from "./createLinePrefixStream"; // Adjust the import path as necessary
import { Transform } from "node:stream";
import { test, expect, vi } from "vitest";
import fs from "fs/promises";
import path from "path";
import os from "os";

test("prefixes every logical line exactly once", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const input = "line1\nline2\nline3";
  const expected = "prefixline1\nprefixline2\nprefixline3\n";

  const inputStream = fs.createReadStream(path.join(dir, "input.txt"), { encoding: "utf8" });
  inputStream.write(input);
  inputStream.end();

  const outputStream = fs.createWriteStream(path.join(dir, "output.txt"));
  const prefixStream = createLinePrefixStream("prefix");
  outputStream.pipe(prefixStream).pipe(outputStream);

  await new Promise((resolve) => outputStream.on("finish", resolve));

  const output = await fs.promises.readFile(path.join(dir, "output.txt"), "utf8");
  expect(output).toBe(expected);
});

test("flushes the final line cleanly when the input ends without a trailing newline", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const input = "line1\nline2";
  const expected = "prefixline1\nprefixline2\n";

  const inputStream = fs.createReadStream(path.join(dir, "input.txt"), { encoding: "utf8" });
  inputStream.write(input);
  inputStream.end();

  const outputStream = fs.createWriteStream(path.join(dir, "output.txt"));
  const prefixStream = createLinePrefixStream("prefix");
  outputStream.pipe(prefixStream).pipe(outputStream);

  await new Promise((resolve) => outputStream.on("finish", resolve));

  const output = await fs.promises.readFile(path.join(dir, "output.txt"), "utf8");
  expect(output).toBe(expected);
});

test("emits nothing for a 0-byte input", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "st-"));
  const input = "";
  const expected = "";

  const inputStream = fs.createReadStream(path.join(dir, "input.txt"), { encoding: "utf8" });
  inputStream.write(input);
  inputStream.end();

  const outputStream = fs.createWriteStream(path.join(dir, "output.txt"));
```
