probe("empty nodes array", async () => {
  return buildCategoryPath([], "123");
});

probe("target not found", async () => {

  return buildCategoryPath([{ id: "1", parentId: null, name: "Root" }], "1");

    ],
    "2"
  );
});

probe("missing intermediate node", async () => {
  return buildCategoryPath(
    [
      { id: "1", parentId: null, name: "Root" },
      { id: "2", parentId: "1", name: "Child" },
      { id: "3", parentId: "4", name: "Grandchild" },
    ],

  );
});
