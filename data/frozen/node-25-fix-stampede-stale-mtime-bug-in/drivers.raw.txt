probe("basic read", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "pr-"));
  const filePath = path.join(dir, "test.txt");
  await fs.promises.writeFile(filePath, "Hello, world!", "utf8");
  const cache = createFileCache({ maxBytes: 1024 });
  const result = await cache.read(filePath);
  await fs.promises.unlink(filePath);
  await fs.promises.rmdir(dir);
  return result.toString();
});

probe("concurrent reads", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "pr-"));
  const filePath = path.join(dir, "test.txt");
  await fs.promises.writeFile(filePath, "Hello, world!", "utf8");
  const cache = createFileCache({ maxBytes: 1024 });
  const [result1, result2] = await Promise.all([
    cache.read(filePath),
    cache.read(filePath),
  ]);
  await fs.promises.unlink(filePath);
  await fs.promises.rmdir(dir);
  return [result1.toString(), result2.toString()];
});

probe("mtimeMs change", async () => {
  const dir = await fs.promises.mkdtemp(path.join(os.tmpdir(), "pr-"));
  const filePath = path.join(dir, "test.txt");

  await fs.promises.utimes(filePath, new Date(), new Date(Date.now() + 1000));
  const result = await cache.read(filePath);
  await fs.promises.unlink(filePath);
  await fs.promises.rmdir(dir);
  return result.toString();
});
