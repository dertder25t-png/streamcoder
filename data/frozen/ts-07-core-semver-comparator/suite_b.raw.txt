- Numeric segments compare numerically rather than lexicographically (so "1.10.0" is greater than "1.2.9").
- "v2.0.0" and "2.0.0" return 0.
- Malformed strings such as "1.2", "1.02.0" (leading zero), or "1.2.x" throw an Error.

```ts
import { compareSemverCore } from './compareSemverCore';

test("Numeric segments compare numerically", () => {
  expect(compareSemverCore("1.10.0", "1.2.9")).toBe(1);
  expect(compareSemverCore("1.2.9", "1.10.0")).toBe(-1);
});

test("'v2.0.0' and '2.0.0' return 0", () => {
  expect(compareSemverCore("v2.0.0", "2.0.0")).toBe(0);
  expect(compareSemverCore("2.0.0", "v2.0.0")).toBe(0);
});

test("Malformed strings throw an Error", () => {
  expect(() => compareSemverCore("1.2", "1.2.0")).toThrow(Error);
  expect(() => compareSemverCore("1.02.0", "1.2.0")).toThrow(Error);
  expect(() => compareSemverCore("1.2.x", "1.2.0")).toThrow(Error);
});
```
